cmake_minimum_required(VERSION 3.20)
project(HitmanCoopNativeFixtures C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/utf-8)
get_filename_component(HITMAN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PLUGIN "${HITMAN_ROOT}/plugin/Src")
set(TESTS "${HITMAN_ROOT}/core/tests")
enable_testing()
add_subdirectory("${HITMAN_ROOT}/third_party/minhook" "${CMAKE_CURRENT_BINARY_DIR}/minhook")

add_executable(coop_pose_tests "${TESTS}/Main.cpp" "${HITMAN_ROOT}/tools/TestPlayerPoses.cpp" "${PLUGIN}/PoseHook.cpp")
add_executable(coop_guard_tests "${TESTS}/Main.cpp" "${HITMAN_ROOT}/tools/TestPlayerGuards.cpp" "${PLUGIN}/CrashGuards.cpp" "${PLUGIN}/NpcReasoningFence.cpp" "${PLUGIN}/NpcMutationGate.cpp")
add_executable(coop_npc_native_tests "${TESTS}/Main.cpp" "${HITMAN_ROOT}/tools/TestNpcReasoningFence.cpp" "${HITMAN_ROOT}/tools/TestNpcMutationGate.cpp" "${HITMAN_ROOT}/tools/TestNpcLifeReceipts.cpp" "${PLUGIN}/CrashGuards.cpp" "${PLUGIN}/NpcReasoningFence.cpp" "${PLUGIN}/NpcMutationGate.cpp" "${PLUGIN}/NpcLifeReceipts.cpp")
target_compile_definitions(coop_guard_tests PRIVATE COOP_GUARD_TESTS)
target_compile_definitions(coop_npc_native_tests PRIVATE COOP_GUARD_TESTS)
target_sources(coop_npc_native_tests PRIVATE "${HITMAN_ROOT}/tools/TestNpcActionProbe.cpp" "${PLUGIN}/NpcActionProbe.cpp")
target_sources(coop_npc_native_tests PRIVATE "${HITMAN_ROOT}/tools/TestNpcActorBirths.cpp" "${PLUGIN}/NpcActorBirths.cpp")
target_sources(coop_npc_native_tests PRIVATE "${HITMAN_ROOT}/tools/TestNpcBehaviorReadback.cpp" "${PLUGIN}/NpcBehaviorReadback.cpp")
add_executable(coop_npc_phase_tests "${TESTS}/Main.cpp" "${HITMAN_ROOT}/tools/TestNpcActionPhase.cpp" "${PLUGIN}/NpcActionPhase.cpp" "${PLUGIN}/NpcActionProbe.cpp" "${PLUGIN}/NpcMutationGate.cpp")
target_sources(coop_npc_phase_tests PRIVATE "${PLUGIN}/NpcBehaviorReadback.cpp")
foreach(target coop_pose_tests coop_guard_tests coop_npc_native_tests coop_npc_phase_tests)
    target_include_directories(${target} PRIVATE "${PLUGIN}" "${TESTS}" "${HITMAN_ROOT}/core/include")
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 30)
endforeach()
add_executable(coop_npc_site_inspector "${HITMAN_ROOT}/tools/InspectNpcSites.cpp" "${PLUGIN}/CrashGuards.cpp" "${PLUGIN}/NpcReasoningFence.cpp" "${PLUGIN}/NpcMutationGate.cpp")
target_include_directories(coop_npc_site_inspector PRIVATE "${PLUGIN}")
target_compile_definitions(coop_npc_site_inspector PRIVATE COOP_GUARD_TESTS)
add_executable(coop_npc_observer_tests "${TESTS}/Main.cpp" "${HITMAN_ROOT}/tools/TestNpcActionObserver.cpp" "${PLUGIN}/NpcActionObserver.cpp" "${PLUGIN}/NpcBehaviorReadback.cpp")
target_include_directories(coop_npc_observer_tests PRIVATE "${PLUGIN}" "${TESTS}")
target_link_libraries(coop_npc_observer_tests PRIVATE coop_minhook)
foreach(case Lifetime WrongThread InvalidSpan CallbackFault StopDuringNative StopDuringCallback InstallWhileExecuting RefusalDiagnostics)
    add_test(NAME observer_${case} COMMAND coop_npc_observer_tests Observer_${case})
    set_tests_properties(observer_${case} PROPERTIES TIMEOUT 30)
endforeach()
